// ---------- Hero node-graph canvas ----------
(function(){
  const canvas = document.getElementById('hero-canvas');
  const ctx = canvas.getContext('2d');
  const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  let w,h,nodes=[];
  function resize(){
    w = canvas.width = canvas.offsetWidth * devicePixelRatio;
    h = canvas.height = canvas.offsetHeight * devicePixelRatio;
  }
  function init(){
    resize();
    const count = Math.max(18, Math.floor(w/140));
    nodes = Array.from({length:count}, ()=>({
      x: Math.random()*w, y: Math.random()*h,
      vx: (Math.random()-0.5)*0.15, vy:(Math.random()-0.5)*0.15,
      r: Math.random()*1.6+1
    }));
  }
  function step(){
    ctx.clearRect(0,0,w,h);
    for(const n of nodes){
      if(!reduceMotion){ n.x+=n.vx; n.y+=n.vy; }
      if(n.x<0||n.x>w) n.vx*=-1;
      if(n.y<0||n.y>h) n.vy*=-1;
    }
    for(let i=0;i<nodes.length;i++){
      for(let j=i+1;j<nodes.length;j++){
        const a=nodes[i], b=nodes[j];
        const dx=a.x-b.x, dy=a.y-b.y;
        const dist=Math.sqrt(dx*dx+dy*dy);
        const maxDist = 220*devicePixelRatio;
        if(dist<maxDist){
          ctx.strokeStyle = `rgba(87,220,198,${0.14*(1-dist/maxDist)})`;
          ctx.lineWidth=1;
          ctx.beginPath(); ctx.moveTo(a.x,a.y); ctx.lineTo(b.x,b.y); ctx.stroke();
        }
      }
    }
    for(const n of nodes){
      ctx.beginPath();
      ctx.arc(n.x,n.y,n.r*devicePixelRatio,0,Math.PI*2);
      ctx.fillStyle='rgba(87,220,198,0.55)';
      ctx.fill();
    }
    if(!reduceMotion) requestAnimationFrame(step);
  }
  window.addEventListener('resize', ()=>{ init(); });
  init();
  step();
})();

// ---------- Network trace spine ----------
(function(){
  const svg = document.getElementById('trace-svg');
  if(!svg) return;
  const sections = [
    {id:'hero', label:'01 · ENTRY'},
    {id:'solution', label:'02 · PROBLEM'},
    {id:'segments', label:'03 · SEGMENTS'},
    {id:'architecture', label:'04 · ENGINE'},
    {id:'roadmap', label:'05 · ROADMAP'},
    {id:'contact', label:'06 · CONTACT'}
  ];
  function build(){
    const totalH = document.body.scrollHeight;
    svg.setAttribute('viewBox', `0 0 2 ${totalH}`);
    svg.innerHTML = `<line x1="1" y1="0" x2="1" y2="${totalH}" stroke="#232B38" stroke-width="1"/>`;
    sections.forEach(s=>{
      const el = document.getElementById(s.id);
      if(!el) return;
      const top = el.offsetTop;
      s._y = top;
      const node = document.createElementNS('http://www.w3.org/2000/svg','circle');
      node.setAttribute('cx','1'); node.setAttribute('cy', top); node.setAttribute('r','5');
      node.setAttribute('fill','#0F141C'); node.setAttribute('stroke','#5B6678');
      node.setAttribute('data-id', s.id);
      node.setAttribute('class','trace-c');
      svg.appendChild(node);
    });
  }
  function updateLit(){
    const scrollY = window.scrollY + window.innerHeight*0.4;
    document.querySelectorAll('.trace-c').forEach(c=>{
      const id = c.getAttribute('data-id');
      const s = sections.find(x=>x.id===id);
      if(s && scrollY >= s._y){
        c.setAttribute('fill','#57DCC6'); c.setAttribute('stroke','#57DCC6');
      } else {
        c.setAttribute('fill','#0F141C'); c.setAttribute('stroke','#5B6678');
      }
    });
  }
  window.addEventListener('load', ()=>{ build(); updateLit(); });
  window.addEventListener('resize', ()=>{ build(); updateLit(); });
  window.addEventListener('scroll', updateLit, {passive:true});
})();
